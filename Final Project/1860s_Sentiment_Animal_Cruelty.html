#Sentiment analysis here the text a once again edited with the stopwords, and then joined together as to get a overall sentiment score from the articles.

fulltext_tidy_recordID <- Dyr_1860s_tidy %>% 
  anti_join(stopwords_1800s, by = c("word" = "word")) %>% 
  group_by(recordID) %>%  
  summarise(fulltext_clean = paste(word, collapse = " "), .groups = "drop")

sentiment_text_recordID <- fulltext_tidy_recordID %>% 
  mutate(
    fulltext_sentiment = map_dbl(
      fulltext_clean,
      function(x) get_sentiment(x, method = "afinn")
    )
  )
sentiment_by_record_1860s <- sentiment_text_recordID %>%
  group_by(recordID) %>%
  summarise(
    mean_sentiment = mean(fulltext_sentiment, na.rm = TRUE),
    n = n()
  )

#Visualizing the Sentiment here the articles sentiment score is plotted into a histogram to better visualize the spread of sentiment across all the articles

library(ggplot2)

#Categorizing the sentiment by dividing all the articles into 3 Categories, “Positiv”, “Negativ” and “Neutral” and counting how many of the articles land into each categories

sentiment_CAT_1860s <- sentiment_by_record_1860s %>%
  mutate(sentiment_category = case_when(
    mean_sentiment > 0 ~ "Positiv",
    mean_sentiment < 0 ~ "Negativ",
    TRUE ~ "Neutral"
  ))
